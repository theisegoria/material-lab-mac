cmake_minimum_required(VERSION 3.25)
# Optimised by default; CoreTests #undef NDEBUG so its asserts stay live in Release.
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
project(PBRMaterialLab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(PBRMaterialLab STATIC src/PBRMaterialLab.cpp src/MaterialPresets.cpp)
target_include_directories(PBRMaterialLab PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_options(PBRMaterialLab PRIVATE -Wall -Wextra -Werror)
if(APPLE)
  enable_language(OBJCXX)
  set(CMAKE_OBJCXX_STANDARD 23)
  set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
  find_package(ZLIB REQUIRED)
  add_executable(material-lab src/CLI.mm src/PNGWriter.cpp src/tools/Common.mm src/tools/Inspect.mm src/tools/Export.mm src/tools/ExportAPI.mm src/tools/MCPServer.mm)
  target_link_libraries(material-lab PRIVATE PBRMaterialLab ZLIB::ZLIB "-framework Foundation" "-framework CoreGraphics" "-framework ImageIO" "-framework Accelerate" "-framework CoreML" "-framework CoreVideo" "-framework CoreText")
  target_compile_options(material-lab PRIVATE -fobjc-arc -Wall -Wextra)
endif()
add_executable(pbrlab_core_tests tests/CoreTests.cpp)
target_link_libraries(pbrlab_core_tests PRIVATE PBRMaterialLab)
enable_testing()
add_test(NAME material_lab_core COMMAND pbrlab_core_tests)

if(APPLE)
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
  add_test(NAME material_lab_cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/CLISmoke.py $<TARGET_FILE:material-lab>)
  add_test(NAME material_lab_mcp COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/MCPSmoke.py $<TARGET_FILE:material-lab>)
endif()
endif()
